
from fastapi import FastAPI
import pickle
import numpy as np


app = FastAPI()


with open('model/house_model.pkl', 'rb') as f:
model = pickle.load(f)


@app.get('/')
def home():
return {'message': 'API Prediksi Harga Rumah'}


@app.post('/predict')
def predict(features: dict):
data = np.array(list(features.values())).reshape(1, -1)
prediction = model.predict(data)
return {'predicted_price': prediction[0]}
