<body>
    <h2>Prediksi Harga Rumah</h2>

    <label>Kamar Tidur:</label><br>
    <input type="number" id="bedrooms"><br><br>

    <label>Kamar Mandi:</label><br>
    <input type="number" step="0.1" id="bathrooms"><br><br>

    <label>Luas Rumah (sqft):</label><br>
    <input type="number" id="sqft_living"><br><br>

    <label>Jumlah Lantai:</label><br>
    <input type="number" step="0.1" id="floors"><br><br>

    <button onclick="predict()">Prediksi</button>

    <h3 id="result"></h3>

    <script>
        async function predict() {
            const data = {
                bedrooms: parseInt(document.getElementById("bedrooms").value),
                bathrooms: parseFloat(document.getElementById("bathrooms").value),
                sqft_living: parseInt(document.getElementById("sqft_living").value),
                floors: parseFloat(document.getElementById("floors").value)
            };

            const response = await fetch("http://127.0.0.1:8000/predict", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify(data)
            });

            const result = await response.json();
            document.getElementById("result").innerText =
                "Perkiraan Harga Rumah: Rp " + result.predicted_price.toLocaleString();
        }
    </script>
